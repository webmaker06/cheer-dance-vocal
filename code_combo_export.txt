            <span class="text-sm sm:text-base font-black text-green-700"><span id="val-${crit.id}">${currentVal}</span> / ${crit.max}%</span>
          </div>
          <input type="range" min="0" max="${crit.max}" value="${currentVal}" 
            oninput="updateSliderVal('${crit.id}', this.value)">
        `;
        container.appendChild(card);
      });

      const commentCard = document.createElement("div");
      commentCard.className = "bg-white p-4 rounded-xl border border-slate-200 shadow-sm space-y-2";
      commentCard.innerHTML = `
        <label for="judge-comment" class="block font-bold text-slate-800 text-sm sm:text-base">Judge Remarks / Comments</label>

      `;
      container.appendChild(commentCard);

      calculateJudgeTotal();
    }

    function updateSliderVal(critId, val) {
      document.getElementById(`val-${critId}`).innerText = val;
      calculateJudgeTotal();
    }

    function calculateJudgeTotal() {
      const activeData = getCurrentCompData();
      let total = 0;
      (activeData.criteria || []).forEach(crit => {
        const el = document.getElementById(`val-${crit.id}`);
        if (el) total += Number(el.innerText || 0);
      });
      document.getElementById("judge-total-score").innerText = `${total} pts`;
    }

    function submitScores() {
      const activeData = getCurrentCompData();
      const contestant = document.getElementById("judge-contestant-select").value;
      if (!contestant || !currentJudge) return;

      if (!activeData.scores) activeData.scores = {};
      if (!activeData.scores[contestant]) activeData.scores[contestant] = {};

      const scoreObj = {};
      (activeData.criteria || []).forEach(crit => {
        const el = document.getElementById(`val-${crit.id}`);
        scoreObj[crit.id] = Number(el ? el.innerText : 0);
      });

      const commentText = document.getElementById("judge-comment") ? document.getElementById("judge-comment").value.trim() : "";

      activeData.scores[contestant][currentJudge] = {
        criteria: scoreObj,
        comment: commentText
      };

      syncToFirebase();
      alert(`Scores and comments submitted for ${contestant} in ${selectedCompetition}!`);
    }

    function getProcessedLeaderboardData() {
      const activeData = getCurrentCompData();
      return (activeData.contestants || []).map(c => {
        const judgeData = (activeData.scores && activeData.scores[c]) || {};
        let grandTotal = 0;
        let judgeCount = 0;
        let breakdownHTML = [];
        let breakdownText = [];

        (activeData.judges || []).forEach(j => {
          const jEntry = judgeData[j];
          if (jEntry) {
            const jScores = jEntry.criteria || jEntry;
            const jSum = Object.values(jScores).filter(v => typeof v === 'number').reduce((a, b) => a + b, 0);
            grandTotal += jSum;
            judgeCount++;
            const noteHTML = jEntry.comment ? `<br/><span class="text-slate-500 italic">"${jEntry.comment}"</span>` : '';
            const noteText = jEntry.comment ? ` ("${jEntry.comment}")` : '';
            breakdownHTML.push(`<div><strong>${j}:</strong> ${jSum}pts${noteHTML}</div>`);
            breakdownText.push(`${j}: ${jSum}pts${noteText}`);
          } else {
            breakdownHTML.push(`<div><strong>${j}:</strong> -</div>`);
            breakdownText.push(`${j}: -`);
          }
        });

        const avg = judgeCount > 0 ? (grandTotal / judgeCount).toFixed(1) : "0.0";
        return { 
          contestant: c, 
          grandTotal, 
          avg, 
          breakdownHTML: breakdownHTML.join(""),
          breakdownText: breakdownText.join(" | ")
        };
      }).sort((a, b) => b.avg - a.avg);
    }

    function renderLeaderboard() {
      const tbody = document.getElementById("leaderboard-body");
      tbody.innerHTML = "";

      const rows = getProcessedLeaderboardData();

      rows.forEach((r, idx) => {
        const tr = document.createElement("tr");
        tr.className = idx % 2 === 0 ? "bg-white" : "bg-slate-50/50";
        tr.innerHTML = `
          <td class="p-3 font-bold ${idx === 0 ? 'text-amber-600' : 'text-slate-600'}">#${idx + 1}</td>
          <td class="p-3 font-bold text-slate-800">${r.contestant}</td>
          <td class="p-3 text-xs text-slate-700 space-y-1">${r.breakdownHTML}</td>
          <td class="p-3 text-right font-bold text-green-700">${r.avg}%</td>
          <td class="p-3 text-right font-bold text-slate-800">${r.grandTotal}</td>
        `;
        tbody.appendChild(tr);
      });
    }

    function exportToExcel() {
      const rowsData = getProcessedLeaderboardData();
      const activeData = getCurrentCompData();

      const excelRows = [
        [activeData.title || selectedCompetition],
        ["Event Category:", selectedCompetition],
        [],
        ["Rank", "Contestant", "Judge Breakdown & Remarks", "Average (%)", "Total Points"]
      ];

      rowsData.forEach((r, idx) => {
        excelRows.push([
          `#${idx + 1}`,
          r.contestant,
          r.breakdownText,
          `${r.avg}%`,
          r.grandTotal
        ]);
      });

      const worksheet = XLSX.utils.aoa_to_sheet(excelRows);
      const workbook = XLSX.utils.book_new();
      XLSX.utils.book_append_sheet(workbook, worksheet, "Leaderboard");

      const fileName = `${selectedCompetition.replace(/\s+/g, '_')}_Leaderboard.xlsx`;
      XLSX.writeFile(workbook, fileName);
    }
